<script>
  let tasks       = [];
  let currentIdx  = 0;
  let totalCopied = 0;

  // ── Phase switcher ────────────────────────────────────────
  function showPhase(phase) {
    document.getElementById('inputWrap').style.display    = phase === 'input'    ? 'block' : 'none';
    document.getElementById('progressWrap').style.display = phase === 'progress' ? 'block' : 'none';
    document.getElementById('doneWrap').style.display     = phase === 'done'     ? 'block' : 'none';
  }

  function showDone(msg) {
    document.getElementById('doneMsg').innerText = msg;
    showPhase('done');
  }

  // ── Step 1: Validate input and kick off scan ──────────────
  function startCollect() {
    const link  = document.getElementById('folderLink').value.trim();
    const errEl = document.getElementById('errorMsg');

    if (!link) {
      errEl.innerText = '⚠ Please enter a folder link.';
      return;
    }
    if (!link.includes('drive.google.com')) {
      errEl.innerText = '⚠ Not a valid Google Drive URL.';
      return;
    }

    errEl.innerText = '';
    document.getElementById('startBtn').disabled = true;
    showPhase('progress');
    document.getElementById('progressStatus').innerText = 'Scanning sub-folders…';

    google.script.run
      .withSuccessHandler(gotTasks)
      .withFailureHandler(handleError)
      .getCollectTasks(link);
  }

  // ── Step 2: Receive task list ─────────────────────────────
  function gotTasks(result) {
    if (result && result.error) {
      showPhase('input');
      document.getElementById('startBtn').disabled = false;
      document.getElementById('errorMsg').innerText = '⚠ ' + result.error;
      return;
    }

    tasks       = result.tasks || [];
    currentIdx  = 0;
    totalCopied = 0;

    if (tasks.length === 0) {
      showDone('No slides found to collect.');
      return;
    }

    updateBar();
    processNext();
  }

  // ── Step 3: Process tasks one by one ─────────────────────
  function processNext() {
    if (currentIdx >= tasks.length) {
      // All done — call finalize then show success
      google.script.run
        .withSuccessHandler(() =>
          showDone(
            'Collected ' + totalCopied + ' slide(s) from ' + tasks.length + ' file(s).'
          )
        )
        .finalizeTasks('collect');
      return;
    }

    const task = tasks[currentIdx];
    document.getElementById('progressStatus').innerText = '📂 ' + task.folderName;
    document.getElementById('progressSub').innerText    = task.fileName;

    google.script.run
      .withSuccessHandler(onTaskDone)
      .withFailureHandler(onTaskError)
      .executeCollectTask(task);
  }

  function onTaskDone(result) {
    if (result && result.copied) totalCopied += result.copied;
    currentIdx++;
    updateBar();
    processNext();
  }

  function onTaskError(err) {
    // Skip failed task and continue
    console.error('Task error:', err);
    currentIdx++;
    updateBar();
    processNext();
  }

  function handleError(err) {
    showPhase('input');
    document.getElementById('startBtn').disabled = false;
    document.getElementById('errorMsg').innerText = '⚠ ' + (err.message || err);
  }

  // ── UI helper ─────────────────────────────────────────────
  function updateBar() {
    const pct = tasks.length > 0 ? Math.round((currentIdx / tasks.length) * 100) : 0;
    document.getElementById('progressFill').style.width = pct + '%';
    document.getElementById('progressPct').innerText    = pct + '%';
  }
</script>
