<script>
  let mode = "<?= mode ?>"; // 'sort' or 'push'
  let tasks = [];
  let currentIndex = 0;

  function init() {
    document.getElementById("taskLabel").innerText = (mode === 'push') ? "Pushing Slides..." : "Sorting Slides...";
    
    // Request list of tasks from backend
    if (mode === 'push') {
      google.script.run.withSuccessHandler(startTasks).getPushTasks();
    } else if (mode === 'sort') {
      google.script.run.withSuccessHandler(startTasks).getSortTasks();
    }
  }

  function startTasks(backendTasks) {
    tasks = backendTasks;
    currentIndex = 0;
    
    if (tasks.length === 0) {
      showDone("No matching items found. Everything is up to date.");
      return;
    }

    updateUI();
    processNext();
  }

  function processNext() {
    if (currentIndex >= tasks.length) {
      google.script.run.withSuccessHandler(() => {
        showDone("Successfully processed " + tasks.length + (mode === 'push' ? " vendors." : " items."));
      }).finalizeTasks(mode);
      return;
    }

    let task = tasks[currentIndex];
    
    if (mode === 'push') {
      document.getElementById("progressStatus").innerText = "Pushing vendor: " + task.vendor;
      google.script.run
        .withSuccessHandler(taskSuccess)
        .withFailureHandler(taskFailure)
        .executePushTask(task);
    } else if (mode === 'sort') {
      document.getElementById("progressStatus").innerText = "Organizing slide order... please wait";
      google.script.run
        .withSuccessHandler(taskSuccess)
        .withFailureHandler(taskFailure)
        .executeSortTask(task);
    }
  }

  function taskSuccess() {
    currentIndex++;
    updateUI();
    processNext();
  }

  function taskFailure(error) {
    console.error("Task failed: " + error);
    // Continue despite error
    currentIndex++;
    updateUI();
    processNext();
  }

  function updateUI() {
    let pct = (tasks.length > 0) ? Math.round((currentIndex / tasks.length) * 100) : 0;
    document.getElementById("progressFill").style.width = pct + "%";
    document.getElementById("progressPct").innerText = pct + "%";
  }

  function showDone(msg) {
    document.getElementById("progressWrap").style.display = "none";
    let doneWrap = document.getElementById("doneWrap");
    doneWrap.style.display = "block";
    document.getElementById("doneMsg").innerText = msg;
  }

  // Start process on load
  window.onload = init;
</script>
